import nltk
import string
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
from nltk.stem import WordNetLemmatizer
from collections import Counter
import math

    # Convert text to lowercase

# Compute Term Frequency (TF)
def compute_tf(doc):
    bow = Counter(doc)  # Word frequencies in document
    doc_len = len(doc)  # Total words in document
    return {word: count / doc_len for word, count in bow.items()}  # Normalized frequency

# Compute Inverse Document Frequency (IDF)
def compute_idf(corpus):
    idf_dict = {}
    total_docs = len(corpus)  # Number of documents
    all_words = set([word for doc in corpus for word in doc])  # Unique words in corpus
    
    for word in all_words:
        containing_docs = sum(1 for doc in corpus if word in doc)  # Docs containing the word
        idf_dict[word] = math.log((total_docs + 1) / (containing_docs + 1)) + 1  # Smoothed IDF
    return idf_dict

# Compute TF-IDF for corpus
def compute_tfidf(corpus):
    idf = compute_idf(corpus)  # Get IDF values
    tfidf_corpus = []
    for doc in corpus:
        tf = compute_tf(doc)  # Compute TF for document
        tfidf = {word: tf[word] * idf[word] for word in doc}  # TF-IDF = TF * IDF
        tfidf_corpus.append(tfidf)
    return tfidf_corpus

# Generate TF-IDF vectors
tfidf_vectors = compute_tfidf(corpus)

"""
Purpose:
This program computes and displays a statistical summary for numerical data.
It calculates mean, median, mode, standard deviation, variance, minimum, 
maximum, and count of valid values. Missing values are handled automatically.
"""

import numpy as np
import pandas as pd
from scipy import stats


# === Function to compute statistical summary ===
def summarize_numerical_data(data):
    # Convert input list to NumPy array for numerical operations
    data = np.array(data)
    
    # Remove NaN values if any
    data = data[~np.isnan(data)]

    # Calculate mean (average value)
    mean_val = np.mean(data)
    
    # Calculate median (middle value)
    median_val = np.median(data)
    
    # Calculate mode (most frequent value) – may return multiple modes
    mode_res = stats.mode(data, keepdims=True)
    
    # Calculate standard deviation (spread of values) – population formula
    std_dev = np.std(data, ddof=0)
    
    # Calculate variance (square of standard deviation)
    variance = np.var(data, ddof=0)
    
    # Find minimum value
    min_val = np.min(data)
    
    # Find maximum value
    max_val = np.max(data)

    # Display all computed statistics
    print("=== Statistical Summary ===")
    print(f"Count     : {len(data)}")
    print(f"Mean      : {mean_val:.2f}")
    print(f"Median    : {median_val:.2f}")
    print(f"Mode      : {mode_res.mode[0]} (Count: {mode_res.count[0]})")
    print(f"Std. Dev  : {std_dev:.2f}")
    print(f"Variance  : {variance:.2f}")
    print(f"Min       : {min_val}")
    print(f"Max       : {max_val}")

"""
Purpose:
This program demonstrates hypothesis testing in different real-world scenarios using SciPy.
It covers:
1. A/B Testing → Comparing two independent groups (e.g., clicks).
2. Drug Trial → Paired t-test for before/after treatment in the same group.
3. Survey Analysis → Independent t-test between two groups of survey responses.
"""

import numpy as np
from scipy import stats


# === 1. A/B Test: Is A better than B? ===
def ab_test_clicks(clicks_A, clicks_B, alpha=0.05):
    """
    Performs an independent t-test (Welch’s test) between group A and B clicks.
    """
    # Run Welch’s t-test (assumes unequal variances)
    t_stat, p_val = stats.ttest_ind(clicks_A, clicks_B, equal_var=False)

    # Display test results
    print("\n--- A/B Test Results ---")
    print(f"T-statistic: {t_stat:.3f}")
    print(f"P-value    : {p_val:.4f}")

    # Decision rule: reject null if p < alpha
    if p_val < alpha:
        print("✅ Result: Statistically significant — A is different from B.")
    else:
        print("❌ Result: No significant difference between A and B.")


# === 2. Drug Trial: Did the drug improve results? ===
def drug_trial(before, after, alpha=0.05):
    """
    Performs a paired t-test (same patients before and after treatment).
    """
    # Paired t-test (compares before vs after for same individuals)
    t_stat, p_val = stats.ttest_rel(before, after)

    # Display test results
    print("\n--- Drug Trial Results ---")
    print(f"T-statistic: {t_stat:.3f}")
    print(f"P-value    : {p_val:.4f}")

    # Decision rule
    if p_val < alpha:
        print("✅ Result: Statistically significant improvement after treatment.")
    else:
        print("❌ Result: No significant improvement.")


# === 3. Survey: Are there differences between groups? ===
def survey_analysis(group1, group2, alpha=0.05):
    """
    Tests if survey responses differ significantly between two groups.
    """
    # Independent samples t-test
    t_stat, p_val = stats.ttest_ind(group1, group2, equal_var=False)

    # Display test results
    print("\n--- Survey Analysis ---")
    print(f"T-statistic: {t_stat:.3f}")
    print(f"P-value    : {p_val:.4f}")

    # Decision rule
    if p_val < alpha:
        print("✅ Result: Groups differ significantly.")
    else:
        print("❌ Result: No significant difference between groups.")

"""

1) Content-Based Filtering → Recommends books with similar themes to what the user liked.

# ===== Imports =====
import numpy as np                     # For arrays and math operations
import pandas as pd                    # For handling small tables
from sklearn.metrics.pairwise import cosine_similarity  # For similarity scoring

# Helper function: pretty-print top-N recommendations
def show_top(df, n=3, cols=("item", "score", "reason")):
    print(df.head(n)[list(cols)].to_string(index=False))


# ------------------------------------------------------------------
# 1) BOOKS → Content-Based (match themes)
# ------------------------------------------------------------------
print("\n=== Books: Content-Based Recommendations ===")

# Small catalog of motivational books with simple pipe-separated themes
books = pd.DataFrame({
    "book_id": [1, 2, 3, 4, 5],
    "title": ["Atomic Habits", "Mindset", "Grit", "The 7 Habits", "Deep Work"],
    "themes": ["habits|identity|systems",
               "growth|learning|effort",
               "perseverance|effort|resilience",
               "habits|principles|leadership",
               "focus|deep-work|habits"]
})

user_likes = [1]  # User liked "Atomic Habits"

# Build a list of all unique themes across all books
all_themes = sorted({t for row in books["themes"] for t in row.split("|")})

# One-hot encode: create 0/1 columns for each theme
for t in all_themes:
    books[t] = books["themes"].apply(lambda s: 1 if t in s.split("|") else 0)

# Build a simple user profile by averaging themes of liked books
theme_cols = all_themes
liked_vecs = books[books["book_id"].isin(user_likes)][theme_cols].to_numpy()
user_profile = liked_vecs.mean(axis=0, keepdims=True) if len(liked_vecs) else np.zeros((1, len(theme_cols)))

# Score all books by cosine similarity to the user profile
item_matrix = books[theme_cols].to_numpy()
books["score"] = cosine_similarity(item_matrix, user_profile).ravel()

# Rank books excluding the ones already liked
ranked_books = (books[~books["book_id"].isin(user_likes)]
                .sort_values("score", ascending=False)
                .assign(reason="Similar themes to your liked book")
                .rename(columns={"title": "item"}))

print("Top book suggestions:")
show_top(ranked_books, cols=("item", "score", "reason"))

3) Manufacturing Measurements -> IQR Rule
"""

import numpy as np
import pandas as pd
from sklearn.ensemble import IsolationForest

# ============================================================
# 1) Credit Card Transactions -> Isolation Forest
#    Detects unusual purchase amounts (fraud-like behavior).
# ============================================================
print("\n=== 1) Credit Card Transactions (Isolation Forest) ===")

# Create synthetic dataset: mostly normal amounts + a few big outliers
np.random.seed(42)
normal_amounts = np.random.normal(loc=50, scale=10, size=100)   # normal purchases
fraud_amounts  = np.array([200, 350, 500])                      # suspiciously large
amounts = np.concatenate([normal_amounts, fraud_amounts])
df_tx = pd.DataFrame({"amount": amounts})

# Train Isolation Forest (unsupervised anomaly detection)
iso = IsolationForest(random_state=42, contamination=0.05)  # contamination ~ fraction anomalies
df_tx["anomaly_score"] = iso.fit_predict(df_tx[["amount"]])  # +1 normal, -1 anomaly

# Map numeric predictions to human-friendly labels
df_tx["label"] = df_tx["anomaly_score"].map({1: "Normal", -1: "Anomaly"})

print(df_tx.tail(10))  # show last few rows
print("\nFlagged anomalies (amounts):")
print(df_tx[df_tx["label"] == "Anomaly"][["amount"]].to_string(index=False))


# ============================================================
# 2) Server Logs -> Z-Score Threshold
#    Detects unusual spikes in requests per minute.
# ============================================================
print("\n=== 2) Server Logs (Z-score threshold) ===")

# Fake "requests per minute" with one big spike
reqs = np.array([60, 58, 62, 61, 59, 63, 60, 59, 62, 61, 190, 62, 61, 60])
df_logs = pd.DataFrame({"rpm": reqs})

# Compute mean and standard deviation
mu = df_logs["rpm"].mean()
sigma = df_logs["rpm"].std(ddof=1)

# Z-score: (value - mean) / std deviation
df_logs["z"] = (df_logs["rpm"] - mu) / sigma

# Simple anomaly rule: mark as anomaly if |z| > 3
df_logs["label"] = np.where(np.abs(df_logs["z"]) > 3, "Anomaly", "Normal")

print(df_logs)
print("\nIndices flagged as anomalies:", df_logs.index[df_logs["label"] == "Anomaly"].tolist())


# ============================================================
# 3) Manufacturing -> IQR Rule
#    Detects abnormal measurements outside boxplot whiskers.
# ============================================================
print("\n=== 3) Manufacturing (IQR rule) ===")

# Tiny dataset with length & weight features
data = np.array([
    [100.2,  49.8],
    [99.9,   50.1],
    [100.5,  49.7],
    [100.1,  50.2],
    [98.8,   60.0],   # suspicious weight
    [120.0,  49.9]    # suspicious length
])
df_mfg = pd.DataFrame(data, columns=["length_mm", "weight_g"])

# Compute IQR bounds (Q1 - 1.5*IQR, Q3 + 1.5*IQR)
bounds = {}
for col in df_mfg.columns:
    q1 = df_mfg[col].quantile(0.25)
    q3 = df_mfg[col].quantile(0.75)
    iqr = q3 - q1
    low  = q1 - 1.5 * iqr
    high = q3 + 1.5 * iqr
    bounds[col] = (low, high)

# Function to check if row is anomaly
def is_anomaly_row(row):
    for col in df_mfg.columns:
        low, high = bounds[col]
        if (row[col] < low) or (row[col] > high):
            return True
    return False

# Apply anomaly detection row-wise
df_mfg["label"] = df_mfg.apply(lambda r: "Anomaly" if is_anomaly_row(r) else "Normal", axis=1)

print(df_mfg)
print("\nAnomalies found:")
print(df_mfg[df_mfg["label"] == "Anomaly"].to_string(index=False))

PURPOSE:

- IoT devices → Real-time monitoring (simple rolling-window alerts)
- Retail data → Sales optimization (basic insights + restock suggestions)

NOTE:

- For real "big data", replace the example lists with file/database/stream inputs
  and consider using tools like Spark, Dask, Kafka, etc.
"""

from collections import deque, Counter
from datetime import datetime, timedelta
from typing import Iterable, Dict, List, Tuple

# -------------------------------
# 1) SOCIAL MEDIA → TREND ANALYSIS
# -------------------------------

def analyze_social_trends(posts: List[Dict], keywords: List[str], bucket="day") -> Dict[str, Dict[str, int]]:
    """
    posts: list of {"ts": datetime, "text": str}
    keywords: list of keywords/hashtags to track, e.g., ["#AI", "sale", "new"]
    bucket: "day" or "hour" for time bucketing
    returns: {bucket_key: {keyword: count}}
    """
    # Create a nested dict {time_bucket: Counter({keyword: count})}
    result: Dict[str, Counter] = {}

    for p in posts:
        # Pick the time bucket label
        if bucket == "hour":
            tb = p["ts"].strftime("%Y-%m-%d %H:00")
        else:
            tb = p["ts"].strftime("%Y-%m-%d")

        # Initialize the counter if needed
        if tb not in result:
            result[tb] = Counter()

        text_lower = p["text"].lower()

        # Count each keyword occurrence once per post (simple rule)
        for kw in keywords:
            if kw.lower() in text_lower:
                result[tb][kw] += 1

    # Convert Counters to normal dicts for readability
    return {tb: dict(cnt) for tb, cnt in result.items()}


# --------------------------------
# 2) IoT DEVICES → REAL-TIME MONITOR
# --------------------------------

def monitor_iot_stream(
    readings: Iterable[Tuple[datetime, float]],
    window_seconds: int = 60,
    alert_threshold: float = 75.0
) -> List[str]:
    """
    readings: stream of (timestamp, value), e.g., (datetime, temperature)
    window_seconds: rolling window size for average
    alert_threshold: alert if windowed average exceeds this
    returns: list of alert messages
    """
    # Use deque for an efficient rolling time window
    window = deque()  # holds (ts, value)
    alerts = []
    running_sum = 0.0

    for ts, value in readings:
        # Add the new reading
        window.append((ts, value))
        running_sum += value

        # Remove stale readings outside the window
        cutoff = ts - timedelta(seconds=window_seconds)
        while window and window[0][0] < cutoff:
            old_ts, old_val = window.popleft()
            running_sum -= old_val

        # Compute average safely
        avg = running_sum / max(len(window), 1)

        # Simple alert rule
        if avg > alert_threshold:
            alerts.append(f"[ALERT] {ts.isoformat()} rolling_avg={avg:.2f} > threshold={alert_threshold}")

    return alerts


# -----------------------------------
# 3) RETAIL DATA → SALES OPTIMIZATION
# -----------------------------------

def optimize_retail_sales(sales: List[Dict], inventory: Dict[str, int], ma_window: int = 3) -> Dict:
    """
    sales: list of {"date": datetime, "sku": str, "qty": int}
    inventory: {sku: current_stock}
    ma_window: moving average window (days) per SKU
    returns: dict with top_skus, daily_totals, moving_avg, restock_suggestions
    """
    # Aggregate daily totals (all SKUs)
    daily_totals: Dict[str, int] = {}
    for s in sales:
        day = s["date"].strftime("%Y-%m-%d")
        daily_totals[day] = daily_totals.get(day, 0) + s["qty"]

    # Aggregate per SKU (keep by date for moving average)
    per_sku_by_day: Dict[str, Dict[str, int]] = {}
    for s in sales:
        day = s["date"].strftime("%Y-%m-%d")
        per_sku_by_day.setdefault(s["sku"], {})
        per_sku_by_day[s["sku"]][day] = per_sku_by_day[s["sku"]].get(day, 0) + s["qty"]

    # Compute simple moving average per SKU (by date order)
    moving_avg: Dict[str, float] = {}
    for sku, day_map in per_sku_by_day.items():
        # Sort by date string (YYYY-MM-DD sorts correctly)
        days_sorted = sorted(day_map.keys())
        series = [day_map[d] for d in days_sorted]

        # If not enough points, average whatever we have
        if len(series) < ma_window:
            moving_avg[sku] = sum(series) / max(len(series), 1)
        else:
            # Last MA value is a simple demand proxy
            window_vals = series[-ma_window:]
            moving_avg[sku] = sum(window_vals) / ma_window

    # Top SKUs by total qty sold
    sku_totals: Dict[str, int] = {}
    for s in sales:
        sku_totals[s["sku"]] = sku_totals.get(s["sku"], 0) + s["qty"]
    top_skus = sorted(sku_totals.items(), key=lambda x: x[1], reverse=True)

    # Restock suggestion: if inventory < (next-day demand proxy = MA * 1.2 safety)
    restock_suggestions = []
    for sku, ma in moving_avg.items():
        needed = int(round(ma * 1.2))  # add small safety factor
        current = inventory.get(sku, 0)
        if current < needed:
            restock_suggestions.append({
                "sku": sku,
                "current_stock": current,
                "suggested_reorder_qty": max(needed - current, 1)
            })

    return {
        "top_skus": top_skus,
        "daily_totals": daily_totals,
        "moving_avg": moving_avg,
        "restock_suggestions": restock_suggestions
    }
