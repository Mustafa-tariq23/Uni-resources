Rule-based outlier flag (if with comparisons)¶

In [11]:
# Purpose: This program demonstrates a simple outlier detection rule.
# It checks if the last value in a list is unusually large compared 
# to the median of the previous values.

# A numeric list with one obvious outlier (120)
values = [5, 7, 6, 8, 120, 6]  

# Select the last value in the list for inspection
x = values[-1]                 

# Take all previous values except the last one
prev = values[:-1]

# Calculate the median of the previous values
# (for odd length, middle value after sorting)
median = sorted(prev)[len(prev) // 2]  

# Start with no outlier flag
flag = False

# Rule: mark as outlier if the last value > 3 × median
if x > 3 * median:
    flag = True

# Show the results
print("Previous values:", prev)
print("Last value:", x)
print("Median of previous:", median)
print("Outlier flagged?", flag)

Last value: 6

Outlier flagged? False

Nested loops — term co-occurrence matrix (mini)¶

In [16]:
# Purpose: This program builds a simple word co-occurrence matrix.
# It counts how often words appear next to each other in a sentence.

# Split the text into tokens (words)
tokens = "respect builds trust and respect fosters success".split()

# Build a sorted list of unique words (the vocabulary)
vocab = sorted(set(tokens))

# Initialize a nested dictionary (matrix) with counts set to 0
# Example: matrix["respect"]["trust"] = 0 initially
matrix = {w: {u: 0 for u in vocab} for w in vocab}

# Define the context window (1 = immediate neighbors left/right)
window = 1

# Loop through each word in the token list
for i, w in enumerate(tokens):
    # Look at words within the window (before and after current word)
    for j in range(max(0, i - window), min(len(tokens), i + window + 1)):
        if i != j:  # Skip the word itself
            matrix[w][tokens[j]] += 1  # Increment co-occurrence count

# Print co-occurrence counts for the word "respect"
print("Co-occur count for 'respect':", matrix['respect'])

Co-occur count for 'respect': {'and': 1, 'builds': 1, 'fosters': 1, 'respect': 0, 'success': 0, 'trust': 0}

# Purpose: This program calculates basic word length statistics

def length_stats(tokens):
    # Create a list of word lengths
    lens = [len(w) for w in tokens]
    # Return minimum, maximum, and average length
    return min(lens), max(lens), sum(lens) / len(lens)

# Initialize counters
lines = 0
words = 0
chars = 0

# Open the file in read mode
with open("quotes.txt", "r", encoding="utf-8") as f:
    # Process the file line by line
    for line in f:

# Purpose: This program counts word frequencies in text, 
# saves the result as a JSON file, and then reads it back.

import json

# Split text into words
text = "respect and respect build trust and character".split()

# Count word frequencies using a dictionary
freq = {}
for w in text:
    freq[w] = freq.get(w, 0) + 1

# Save the frequency dictionary into a JSON file
# ensure_ascii=False → keeps text readable (e.g., non-English chars)
# indent=2 → makes the file nicely formatted
with open("freq.json", "w", encoding="utf-8") as f:
    json.dump(freq, f, ensure_ascii=False, indent=2)
print("Saved freq.json")

# Read the JSON file back into a Python dictionary
with open("freq.json", "r", encoding="utf-8") as f:
    loaded = json.load(f)

# Print the loaded dictionary
print(loaded)

# Purpose: This program calculates a simple moving average (SMA) 
# for a sequence of values and visualizes it using matplotlib.

import matplotlib.pyplot as plt

# A list of numeric values
values = [5, 7, 4, 9, 6, 10, 8, 12]

# Window size for moving average
window = 3

# List to store moving average values
ma = []

# Loop through each index in the list
for i in range(len(values)):
    # Define the start of the window (avoid going below index 0)
    start = max(0, i - window + 1)
    # Compute average of values in the current window
    ma.append(sum(values[start:i+1]) / (i - start + 1))

# Plot the original values
plt.plot(values, marker="o")

# Plot the moving average
plt.plot(ma, marker="s")

# Add chart title
plt.title("Values vs Moving Average")

# Add legend to differentiate lines
plt.legend(["values", "moving_avg"])

# Label the axes
plt.xlabel("Index")
plt.ylabel("Value")

# Show the plot
plt.show()

# Purpose: This program classifies words as "short" or "long" 
# based on their length and visualizes the distribution with a pie chart.

import matplotlib.pyplot as plt
from collections import Counter

# Split the sentence into tokens (words)
tokens = "Respect and discipline build character".split()

# Classify words: short (≤4 letters) or long (>4 letters)
classes = ["short" if len(w) <= 4 else "long" for w in tokens]
