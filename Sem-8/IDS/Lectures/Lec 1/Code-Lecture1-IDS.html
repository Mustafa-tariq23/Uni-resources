from sklearn.linear_model import LinearRegression
import pandas as pd

# Motivational Quote
print("'Indeed, Allah does not allow the reward of those who do good to be lost.' – Quran 9:120")

data = {"Hours_Studied": [2, 4, 6, 8],
        "Exam_Score": [50, 60, 65, 80]}

df = pd.DataFrame(data)

X = df[["Hours_Studied"]]
y = df["Exam_Score"]

model = LinearRegression()
model.fit(X, y)

print("Predicted Score (7 hours study):", model.predict([[7]])[0])
